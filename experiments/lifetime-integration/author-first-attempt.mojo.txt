"""Small author-facing two-draft lifetime contract exercise."""

from adapters import Keyed, field
from delivery import QueuedEvent
from moxi import (
    App, Event, Rect, TextInputEvent, CompositionEvent,
    BUTTON_KIND, TEXT_INPUT_VIEW_KIND,
)
from runtime import (
    HandlerHost, Screen, UI, deliver, require, reset, update_model,
)


struct Editor(Pane):
    comptime Context = NoneType
    var name: String
    var text: String
    var increments: Int

    def __init__(out self, name: String, text: String):
        self.name = name
        self.text = text
        self.increments = 0

    def change(mut self, value: String):
        self.text = value

    def increment(mut self):
        self.increments += 1

    def render(self, mut ui: UI[Self, NoneType], context: NoneType):
        ui.label(self.name)
        ui.text_input(self.name, self.text, Self.change)
        ui.button("Increment", Self.increment)


struct Workspace(Screen):
    var editors: Keyed[Editor]

    def __init__(out self):
        self.editors = Keyed[Editor]()
        _ = self.editors.append(Editor("A", "A clean"))
        _ = self.editors.append(Editor("B", "B clean"))

    def replace_a(mut self):
        for i in range(len(self.editors.entries)):
            if self.editors.entries[i].value.name == "A":
                self.editors.entries[i].value.text = "A replacement"
                self.editors.entries[i].value.increments = 40

    def reorder(mut self):
        var first = self.editors.entries[0]
        var second = self.editors.entries[1]
        self.editors.entries[0] = second
        self.editors.entries[1] = first

    def text_for(self, name: String) -> String:
        for entry in self.editors.entries:
            if entry.value.name == name:
                return entry.value.text
        return ""

    def increments_for(self, name: String) -> Int:
        for entry in self.editors.entries:
            if entry.value.name == name:
                return entry.value.increments
        return -1

    def render(self, mut ui: UI[Self]):
        field["editors"](ui, self, NoneType(), key=1)


def control_id(app: App[HandlerHost[Workspace]], kind: Int, ordinal: Int) -> Int:
    var seen = 0
    for control in app.component.ui.controls:
        if control.active and control.kind == kind:
            if seen == ordinal:
                return control.id
            seen += 1
    return -1


def main():
    var app = App(HandlerHost(Workspace()), Rect(0, 0, 640, 900))
    var text_a = control_id(app, TEXT_INPUT_VIEW_KIND, 0)
    var text_b = control_id(app, TEXT_INPUT_VIEW_KIND, 1)
    var button_a = control_id(app, BUTTON_KIND, 0)
    var button_b = control_id(app, BUTTON_KIND, 1)
    require(text_a >= 0 and text_b >= 0 and button_a >= 0 and button_b >= 0, "initial controls exist")

    var a_text = Event(TextInputEvent("A draft", 0, 1000000))
    a_text.set_target(text_a)
    require(deliver(app, app.component.queue(a_text)), "first draft accepts text")
    var b_text = Event(TextInputEvent("B draft", 0, 1000000))
    b_text.set_target(text_b)
    require(deliver(app, app.component.queue(b_text)), "second draft accepts text")
    require(app.component.root.text_for("A") == "A draft", "editor A owns its text")
    require(app.component.root.text_for("B") == "B draft", "editor B owns its text")

    require(deliver(app, app.component.capture(button_a)), "editor A increments")
    require(deliver(app, app.component.capture(button_b)), "editor B increments")
    require(app.component.root.increments_for("A") == 1, "editor A owns its count")
    require(app.component.root.increments_for("B") == 1, "editor B owns its count")

    var ime = Event(CompositionEvent("ime", 1, 2))
    ime.set_target(text_a)
    require(deliver(app, app.component.queue(ime)), "editor A accepts composition")
    var before_reorder = app.component.ui.controls[app.component.ui.index(text_a)].editor
    var a_action = app.component.capture(button_a)

    update_model(app, Workspace.reorder)
    require(app.component.root.editors.entries[0].value.name == "B", "reorder changes display order")
    require(app.component.root.editors.entries[1].value.name == "A", "reorder keeps both editors")
    require(app.component.root.text_for("A") == "A draft", "reorder keeps A text")
    require(app.component.root.text_for("B") == "B draft", "reorder keeps B text")
    require(app.component.root.increments_for("A") == 1, "reorder keeps A count")
    require(app.component.root.increments_for("B") == 1, "reorder keeps B count")
    require(deliver(app, a_action), "captured A action follows its editor")
    require(app.component.root.increments_for("A") == 2, "captured action still targets A")
    require(app.component.root.increments_for("B") == 1, "captured action does not target B")
    var after_reorder = app.component.ui.controls[app.component.ui.index(text_a)].editor
    require(after_reorder.text == before_reorder.text, "reorder preserves editor text state")
    require(after_reorder.cursor == before_reorder.cursor and after_reorder.anchor == before_reorder.anchor, "reorder preserves selection state")
    require(after_reorder.composition == before_reorder.composition, "reorder preserves composition text")

    var replacement_action = app.component.capture(button_a)
    update_model(app, Workspace.replace_a)
    require(app.component.root.text_for("A") == "A replacement", "same document accepts replacement values")
    require(app.component.root.increments_for("A") == 40, "replacement values include the new count")
    require(deliver(app, replacement_action), "captured action updates replacement")
    require(app.component.root.increments_for("A") == 41, "replacement receives captured callback")
    require(app.component.root.text_for("B") == "B draft", "replacement leaves B alone")

    var stale_action = app.component.capture(button_a)
    var stale_text = Event(TextInputEvent("stale", 0, 1000000))
    stale_text.set_target(text_a)
    var stale_packet: QueuedEvent = app.component.queue(stale_text)
    reset(app, Workspace())
    require(app.component.root.text_for("A") == "A clean", "new workspace discards A draft")
    require(app.component.root.text_for("B") == "B clean", "new workspace discards B draft")
    require(app.component.root.increments_for("A") == 0 and app.component.root.increments_for("B") == 0, "new workspace discards counts")
    require(not deliver(app, stale_action), "old callback is rejected after new workspace")
    require(not deliver(app, stale_packet), "old targeted text is rejected after new workspace")
    require(app.component.root.text_for("A") == "A clean", "stale callback cannot change replacement A")
    require(app.component.root.text_for("B") == "B clean", "stale text cannot change replacement B")
    require(app.component.root.increments_for("A") == 0, "stale callback cannot change replacement count")

    var fresh_button = control_id(app, BUTTON_KIND, 0)
    require(deliver(app, app.component.capture(fresh_button)), "fresh workspace accepts fresh callback")
    require(app.component.root.increments_for("A") == 1, "fresh callback targets fresh A")

    var source = App(HandlerHost(Workspace()), Rect(0, 0, 640, 900))
    var source_text_id = control_id(source, TEXT_INPUT_VIEW_KIND, 0)
    var source_button_id = control_id(source, BUTTON_KIND, 0)
    var source_text = Event(TextInputEvent("source pending", 0, 1000000))
    source_text.set_target(source_text_id)
    var source_packet: QueuedEvent = source.component.queue(source_text)
    var source_action = source.component.capture(source_button_id)
    var duplicate = source
    require(control_id(duplicate, BUTTON_KIND, 0) == source_button_id, "duplicate can share numeric control IDs")
    require(not deliver(duplicate, source_action), "duplicate rejects source callback")
    require(not deliver(duplicate, source_packet), "duplicate rejects source text")
    require(duplicate.component.root.text_for("A") == "A clean", "duplicate remains independent")
    require(duplicate.component.root.increments_for("A") == 0, "duplicate rejects source count change")
    require(deliver(app=source, source_action), "source accepts its own callback")
    require(source.component.root.increments_for("A") == 1, "source receives its own callback")
    require(duplicate.component.root.increments_for("A") == 0, "source callback does not affect duplicate")

    print("PASS author-facing two-draft lifetime contract")
